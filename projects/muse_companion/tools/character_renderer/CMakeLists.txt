cmake_minimum_required(VERSION 3.20)
project(muse_character_preview LANGUAGES C)
if(NOT CHARACTER_SOURCE)
    message(FATAL_ERROR "Set CHARACTER_SOURCE to the reviewed Muse generated C file")
endif()
add_executable(character_dump dump.c "${CHARACTER_SOURCE}")
target_include_directories(character_dump PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
set_property(TARGET character_dump PROPERTY C_STANDARD 11)
if(MSVC)
    target_compile_definitions(character_dump PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(character_dump PRIVATE /utf-8 /W3 /WX)
    if(CHARACTER_ASAN)
        target_compile_options(character_dump PRIVATE /fsanitize=address /Zi)
        target_link_options(character_dump PRIVATE /DEBUG)
    endif()
else()
    target_compile_options(character_dump PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(character_dump PRIVATE m)
    if(CHARACTER_ASAN)
        target_compile_options(character_dump PRIVATE -fsanitize=address,undefined)
        target_link_options(character_dump PRIVATE -fsanitize=address,undefined)
    endif()
endif()
